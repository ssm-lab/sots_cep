@startuml
title Activity Diagram

start

' ============================================================
' Stream Partition
' ============================================================
partition "Stream" {
    :Generate raw event;
    :Forward event to Coordinator;
}

' ============================================================
' Coordinator monitors independently (asynchronous lane)
' ============================================================
partition "Coordinator" {
    :Monitor event timing and schedules;
    if ("Gap detected?") then (yes)
        :Activate reconstruction mode;
    else (no)
        :Flag event as directly observed;
    endif
}

' ============================================================
' Reconstructor Partition
' ============================================================
partition "Reconstructor" {
    :Receive incoming event;

    if ("Reconstruction mode active?") then (yes)
        :Apply Predictor to estimate missing value;
        :Compute confidence using ConfidenceModel;
        :Create ReconstructedEvent;
    else (no)
        :Wrap raw event into ReconstructedEvent;
        :Assign confidence = 1.0;
    endif

    :Publish ReconstructedEvent to EventStream;
}

' ============================================================
' CEP Engine Partition
' ============================================================
partition "CEP Engine" {
    :Consume unified reconstructed event flow;
    :Apply CEP patterns and derive higher-level events;
}

stop
@enduml
