@startuml
title Event Processing and Reconstruction Flow

actor Researcher

participant "External System" as Ext
participant "Event Source" as Src
participant "Event Stream" as ES
participant "Coordinator" as Coor
participant "Reconstructor" as Recon
participant "Predictor" as Pred
participant "Event Processor" as CEP

== Data Acquisition ==
Researcher -> Ext: startDataCollection()
Ext -> Src: observe(value, observation_time)
Src -> ES: emitEvent(e_obs: Event)

== Observation Assessment ==
Coor -> Coor: checkExpectedEventTime(source_ID, current_time)

alt Expected event observed
    ES -> Coor: deliverEvent(e_obs: Event)
    Coor -> Coor: updateLastSeen(source_ID, observation_time)

    ES -> Recon: deliverEvent(e_obs: Event)
    Recon -> Pred: update(value)

    ES -> CEP: deliverEvent(e_obs: Event)

else Expected event missing
    Coor -> Recon: requestReconstruction(source_ID, expected_time)
    Recon -> Pred: predict()
    Pred --> Recon: estimate, confidence
    Recon -> ES: emitEvent(e_rec: Event)

    ES -> CEP: deliverEvent(e_rec: Event)
end

== Computational Processing ==
CEP -> CEP: evaluatePatterns(e: Event)
CEP --> Researcher: reportDetectedPattern()
@enduml
